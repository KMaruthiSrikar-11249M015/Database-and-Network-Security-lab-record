-- ── 1. TABLE, SEQUENCE & TRIGGER CREATION ──────────────

CREATE TABLE Student (
    student_id INT PRIMARY KEY,
    name VARCHAR2(50) NOT NULL,
    email VARCHAR2(80) UNIQUE,
    dept_id INT NOT NULL,
    cgpa NUMBER(4,2) DEFAULT 0.0 CHECK (cgpa BETWEEN 0 AND 10),
    join_date DATE DEFAULT SYSDATE
);

-- Sequence to emulate AUTO_INCREMENT functionality
CREATE SEQUENCE student_seq START WITH 1 INCREMENT BY 1;

-- Trigger to automatically insert sequence value on primary key
CREATE OR REPLACE TRIGGER trg_student_id
BEFORE INSERT ON Student
FOR EACH ROW
BEGIN
    IF :NEW.student_id IS NULL THEN
        SELECT student_seq.NEXTVAL INTO :NEW.student_id FROM dual;
    END IF;
END;
/

-- ── 3. DATA VERIFICATION & SEQUENTIAL TRANSACTIONS ────

-- SQL*Plus System layout setups
SET PAGESIZE 60;
SET LINESIZE 120;
COLUMN name FORMAT A12;
COLUMN email FORMAT A18;

-- Step A: Initial Select Verification
SELECT student_id, name, dept_id, cgpa FROM Student;

-- Step B: Update and Check Row 
UPDATE Student SET cgpa=9.0 WHERE name='Ravi M';
SELECT name, cgpa FROM Student WHERE name='Ravi M';

-- Step C: Delete Verification
DELETE FROM Student WHERE student_id=5;
SELECT student_id, name FROM Student;

-- Step D: Committed Transaction Block
-- In Oracle, transactions start automatically on the first DML statement
INSERT INTO Student (name, email, dept_id, cgpa) VALUES ('Test', 'test@mail.com', 1, 5.0);
COMMIT;
SELECT name, cgpa FROM Student WHERE name='Test';

-- Step E: Rollback Transaction Block
UPDATE Student SET cgpa=0.0 WHERE dept_id=1;
ROLLBACK;
SELECT name, cgpa FROM Student WHERE dept_id=1;
